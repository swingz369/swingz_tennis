# PaginationNav

Reusable pagination navigation component for SwingZ admin pages.

**Location:** `components/ui/pagination-nav.tsx`

---

## Navigation Modes

### 1. Link-based (`buildUrl`) — SSR-friendly

Renders `<a>` tags via Next.js `<Link>`. Best for Server Components or pages where pagination is driven by URL search params (`?page=2`).

```tsx
import { PaginationNav } from '@/components/ui/pagination-nav';
import { getPagination, buildPaginationMeta, buildPageUrl } from '@/lib/pagination';

// In a Server Component (page.tsx):
export default async function MembersPage({ searchParams }) {
  const params = await searchParams;
  const { page, offset, limit } = getPagination(params, 25);

  const { data, count } = await supabase
    .from('members')
    .select('*', { count: 'exact' })
    .range(offset, offset + limit - 1);

  const pagination = buildPaginationMeta(page, limit, count);
  const buildUrl = (p: number) => `?${buildPageUrl(params, p)}`;

  return (
    <div>
      {/* ... member list ... */}
      <PaginationNav meta={pagination} buildUrl={buildUrl} />
    </div>
  );
}
```

**Pages using this mode:** members, billing, audit-logs

---

### 2. onClick-based (`onPageChange`) — Client-side

Calls a callback function on click. Best for client components using `useState` or `router.push`.

```tsx
'use client';
import { useState } from 'react';
import { useRouter } from 'next/navigation';
import { PaginationNav } from '@/components/ui/pagination-nav';
import type { PaginationMeta } from '@/lib/pagination';

// Option A: setState (e.g. court-types with API pagination)
function CourtTypesList() {
  const [page, setPage] = useState(1);
  const [pagination, setPagination] = useState<PaginationMeta | null>(null);

  // ... fetch data on page change ...

  return (
    <div>
      {/* ... table ... */}
      {pagination && <PaginationNav meta={pagination} compact onPageChange={setPage} />}
    </div>
  );
}

// Option B: router.push (e.g. server-rendered pages from client component)
function SeasonsList({ pagination }: { pagination: PaginationMeta }) {
  const router = useRouter();

  return (
    <PaginationNav
      meta={pagination}
      compact
      onPageChange={(p) => router.push(`/admin/seasons?page=${p}`)}
    />
  );
}
```

**Pages using this mode:** seasons, tournaments, court-types, shop, clubs

## Variants

### Default (full)

Shows: `« ‹ 1 2 [3] 4 5 … 10 › »` with first/last buttons and page numbers.

```tsx
<PaginationNav meta={pagination} buildUrl={buildUrl} />
```

### Compact

Shows only: `‹ 3/10 ›` — no first/last buttons, no page numbers.

```tsx
<PaginationNav meta={pagination} compact onPageChange={setPage} />
```

### With page-size selector

Shows a dropdown to change items per page (e.g. 10, 25, 50).

```tsx
<PaginationNav
  meta={pagination}
  buildUrl={buildUrl}
  pageSizeOptions={[10, 25, 50]}
  currentLimit={limit}
  onPageSizeChange={(size) => `?page=1&limit=${size}`}
/>
```

## Props

| Prop               | Type                       | Default      | Description                            |
| ------------------ | -------------------------- | ------------ | -------------------------------------- |
| `meta`             | `PaginationMeta`           | **required** | From `buildPaginationMeta()`           |
| `buildUrl`         | `(page: number) => string` | —            | Link-based navigation (SSR)            |
| `onPageChange`     | `(page: number) => void`   | —            | onClick-based navigation (client)      |
| `compact`          | `boolean`                  | `false`      | Hide page numbers, show only prev/next |
| `pageSizeOptions`  | `number[]`                 | —            | Show page-size selector                |
| `currentLimit`     | `number`                   | —            | Current page size                      |
| `onPageSizeChange` | `(size: number) => string` | —            | Page-size change handler               |
| `className`        | `string`                   | `''`         | Additional CSS classes                 |

> **Note:** Provide either `buildUrl` or `onPageChange`, not both. If neither is provided, buttons render as disabled.

---

## PaginationMeta

Generated by `buildPaginationMeta()` from `@/lib/pagination`:

```ts
interface PaginationMeta {
  page: number; // Current page (1-based)
  limit: number; // Items per page
  totalCount: number; // Total items across all pages
  totalPages: number; // Total number of pages
  hasNext: boolean; // Whether there is a next page
  hasPrev: boolean; // Whether there is a previous page
  nextPage: number | null;
  prevPage: number | null;
  offset: number; // Supabase .range() offset
}
```
